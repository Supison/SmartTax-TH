สถาปัตยกรรมโค้ด (Frontend & Backend Separation)

Front-end Layer (Client)

จัดการ UI, Event Listeners, Form Inputs, Chart.js และส่ง HTTP Request ไปยัง Backend API 

async function fetchTaxCalculation(payload) {
    const response = await fetch(
        '/api/v1/calculate-tax',
        {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            },
            body: JSON.stringify(payload)
        }
    );

    return await response.json();
}

Back-end Layer (Node.js / Express API)

รับข้อมูลรายได้ ค่าลดหย่อน คำนวณภาษี และส่ง JSON Response กลับมา 

app.post('/api/v1/calculate-tax', (req, res) => {
    const taxData = TaxEngine.calculate(req.body);

    res.json({
        status: 'success',
        data: taxData
    });
});
